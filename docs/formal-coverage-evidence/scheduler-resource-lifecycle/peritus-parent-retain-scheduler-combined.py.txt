files = {}


def retain(source):
    source = Path(source)
    name = source.name
    if source.suffix in {'.rs', '.py', '.md'}:
        name += '.txt'
    elif source.suffix == '.log':
        name = name[:-4] + '.txt'
    if name in files:
        assert digest(files[name]) == digest(source), name
    files[name] = str(source)


for name in ['peritus-sol-scheduler-command-final-evidence-001.sha256',
             'peritus-sol-scheduler-worker-lookup-evidence-002.sha256',
             'peritus-sol-scheduler-perf-aggregate-evidence-20260913T001.sha256']:
    path = Path('/tmp') / name
    for line in path.read_text().splitlines():
        expected_digest, source = line.split('  ', 1)
        assert digest(source) == expected_digest, source
        retain(source)
    retain(path)
for name in ['peritus-sol-scheduler316-independent-source-review-20260912T001.md',
