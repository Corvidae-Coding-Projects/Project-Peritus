import datetime
import hashlib
import json
from pathlib import Path
import shutil

root = Path('/home/doll/Project-Peritus/.worktrees/formal-coverage')
package = Path('crates/foundation/peritus-release-policy')
dest = root / 'docs/formal-coverage-evidence/release-candidate-construction'
sha = lambda path: hashlib.sha256(path.read_bytes()).hexdigest()
manifest = Path('/tmp/peritus-sol-release-candidate-final-full-package.sha256')
assert sha(manifest) == '42cdfd534b84c5f7fa8f1e64aabb8957864343ebcb9336754da4b26806873918'
expected = {}
for line in manifest.read_text().splitlines():
    digest, name = line.split('  ', 1)
    assert name not in expected
    expected[name] = digest
    assert sha(root / name) == digest, name
assert len(expected) == 65
assert {str(p.relative_to(root)) for p in (root / package).rglob('*') if p.is_file()} == expected.keys()

worker_evidence = Path('/tmp/peritus-sol-release-candidate-evidence-final.sha256')
assert sha(worker_evidence) == '3697a0513bda2065b005c118b0a8b9f8ea290f4aad646d6b2e410d91ba42fb71'
files = {}
for line in worker_evidence.read_text().splitlines():
    digest, name = line.split('  ', 1)
    path = Path(name)
    assert sha(path) == digest, name
    retained = path.name.replace('peritus-sol-release-candidate-', 'worker-')
    if retained.endswith('.rs'):
        retained += '.txt'
    elif retained.endswith('.log'):
        retained = retained[:-4] + '.txt'
    assert retained not in files
    files[retained] = str(path)

files.update({
    'independent-review.md': '/tmp/peritus-parent-release-candidate-independent-review.md',
    'worker-evidence-sha256.txt': str(worker_evidence),
    'full-package-sha256.txt': str(manifest),
    'preimage-sha256.txt': '/tmp/peritus-parent-release377-source-exact.sha256',
    'changed-source-sha256.txt': '/tmp/peritus-sol-release-candidate-final-changed-source.sha256',
    'source-diff.patch': '/tmp/peritus-sol-release-candidate-baseline-to-final.patch',
    'qualification-parent.json': '/tmp/peritus-parent-release-candidate-qualification-integrated.json',
    'integration-driver.py.txt': '/tmp/peritus-parent-release-candidate-integrate.py',
    'retention-driver.py.txt': __file__,
})
for label in ['verus', 'tests-default', 'tests-all', 'clippy', 'fmt', 'api', 'layout']:
    files[f'{label}-parent.txt'] = f'/tmp/peritus-parent-release-candidate-{label}-integrated.log'

assert not dest.exists(), dest
dest.mkdir()
for name, source in files.items():
    shutil.copyfile(source, dest / name)
    assert sha(dest / name) == sha(Path(source))

record = {
