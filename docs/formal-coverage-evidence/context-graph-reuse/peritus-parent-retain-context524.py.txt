files = {}

def retain(source):
    source = Path(source)
    name = source.name
    if source.suffix in {'.rs', '.py', '.md'}:
        name += '.txt'
    elif source.suffix == '.log':
        name = name[:-4] + '.txt'
    if name in files:
        assert sha(files[name]) == sha(source), name
    files[name] = str(source)

for evidence in [Path('/tmp/peritus-sol-context-final-evidence-20260913T004133Z.sha256'),
                 Path('/tmp/peritus-sol-context-cycle-fix-final3-evidence.sha256')]:
    for line in evidence.read_text().splitlines():
        digest, source = line.split('  ', 1)
        assert sha(source) == digest, source
        retain(source)
    retain(evidence)
for name in ['peritus-parent-context524-independent-review.md',
             'peritus-parent-context524-integrate.py',
             'peritus-parent-context55-to-524-full-package.sha256',
             'peritus-parent-context55-to-524-changed.sha256',

             'peritus-parent-context-final3-source-evidence-check.json',
             'peritus-parent-context-final3-performance-source-check.json']:
    retain(Path('/tmp') / name)
