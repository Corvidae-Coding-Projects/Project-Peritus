import datetime
import difflib
import hashlib
import json
from pathlib import Path
import shutil

root = Path('/tmp/peritus-parent-obligations-proof.6ib_z8mo')
baseline = Path('/tmp/peritus-sol-obligations-performance-source-01')
shared = Path('/home/doll/Project-Peritus/.worktrees/formal-coverage')
frozen = Path('/tmp/peritus-parent-obligations-contracts-source-01')
package = Path('crates/orchestration/peritus-obligations')
prefix = '/tmp/peritus-parent-obligations-contracts'
sha = lambda path: hashlib.sha256(path.read_bytes()).hexdigest()

def entries(path):
    result = {}
    for line in path.read_text().splitlines():
        digest, name = line.split('  ', 1)
        assert name not in result
        result[name] = digest
    return result

preimage = Path('/tmp/peritus-sol-obligations-performance-full-package.sha256')
manifest = Path(prefix + '-full-package.sha256')
assert sha(preimage) == '0ca196662727368857d6dfbd0386b2edf97da307db9e3e536240192fd4f13fec'
assert sha(manifest) == '548fbb7cf5485c65354c41152cb10fc365d419a77c7740dc566ad0bb00f6dd17'
before, after = entries(preimage), entries(manifest)
assert len(before) == 51 and len(after) == 56 and before.keys() <= after.keys()
for source_root, expected in [(baseline, before), (shared, before), (root, after)]:
    found = {str(p.relative_to(source_root)) for p in (source_root / package).rglob('*') if p.is_file()}
    assert found == expected.keys(), str(source_root)
    for name, digest in expected.items():
        assert sha(source_root / name) == digest, (str(source_root), name)
changed = sorted(name for name in after if before.get(name) != after[name])
assert len(changed) == 8
assert not frozen.exists()
for name in after:
    dest = frozen / name
    dest.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(root / name, dest)
    assert sha(dest) == after[name]
changed_manifest = Path(prefix + '-changed-source.sha256')
changed_manifest.write_text(''.join(f'{after[name]}  {name}\n' for name in changed))
patch = Path(prefix + '-source.patch')
patch.write_text(''.join(''.join(difflib.unified_diff(
    (baseline / name).read_text().splitlines(keepends=True) if name in before else [],
    (frozen / name).read_text().splitlines(keepends=True),
    fromfile='a/' + name if name in before else '/dev/null', tofile='b/' + name)) for name in changed))

dependencies = {}
for dependency in ['crates/foundation/peritus-spec', 'crates/foundation/peritus-types',
                   'crates/orchestration/peritus-run-settlement']:
    names = {str(p.relative_to(root)) for p in (root / dependency).rglob('*') if p.is_file()}
    assert names == {str(p.relative_to(shared)) for p in (shared / dependency).rglob('*') if p.is_file()}
    for name in sorted(names):
        dependencies[name] = sha(root / name)
        assert sha(shared / name) == dependencies[name]
dependency_manifest = Path(prefix + '-direct-dependencies.sha256')
dependency_manifest.write_text(''.join(f'{digest}  {name}\n' for name, digest in sorted(dependencies.items())))

qualification = Path(prefix + '-qualification.json')
record = json.loads(qualification.read_text())
assert record['final_source_files_checked'] == 56
assert len(record['commands']) == 10
assert all(c['exit_code'] == (101 if c['label'].startswith('negative-') else 0) for c in record['commands'])
old = entries(Path(prefix + '-pre-const-full-package.sha256'))
assert [name for name in after if old[name] != after[name]] == [str(package / 'tests/qualification_contracts.rs')]

report = Path(prefix + '-implementation-review.md')

artifacts = [report, manifest, preimage, changed_manifest, patch, dependency_manifest, qualification,
             Path(prefix + '-qualification.py'), Path(prefix + '-complete-qualification.py'),
